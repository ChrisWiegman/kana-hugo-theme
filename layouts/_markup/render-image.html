{{- /* Resolve the image from the page bundle first, then the global assets directory. Anything that can't be resolved (remote URLs, files in static/) is rendered as a plain so the build never fails on a missing resource. Processed images get a with a per preferred format in site.Params.imageFormats. */ -}} {{- $u := urls.Parse .Destination -}} {{- $image := "" -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- $image = or (.Page.Resources.Get $path) (resources.Get $path) -}} {{- end -}} {{- $src := .Destination -}} {{- $width := "" -}} {{- $height := "" -}} {{- $srcSet := "" -}} {{- $sources := "" -}} {{- with $image -}} {{- $src = .RelPermalink -}} {{- if and (eq .ResourceType "image") (ne .MediaType.SubType "svg") -}} {{- $width = .Width -}} {{- $height = .Height -}} {{- $srcSet = partial "image-srcset.html" . -}} {{- if $srcSet }}{{ $sources = partial "image-sources.html" (dict "image" . "sizes" site.Params.imageSizeString) }}{{ end -}} {{- end -}} {{- end -}} {{- if .IsBlock -}} {{- else -}} {{ with $sources }}{{ . | safeHTML }}{{ end }}{{ with $sources }}{{ end }} {{- end -}}
