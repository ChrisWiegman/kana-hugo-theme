{{- /* Resolve the image from the page bundle first, then the global assets directory. Anything that can't be resolved (remote URLs, files in static/) is rendered as a plain so the build never fails on a missing resource. */ -}} {{- $u := urls.Parse .Destination -}} {{- $image := "" -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- $image = or (.Page.Resources.Get $path) (resources.Get $path) -}} {{- end -}} {{- $src := .Destination -}} {{- $width := "" -}} {{- $height := "" -}} {{- $srcSet := "" -}} {{- with $image -}} {{- $src = .RelPermalink -}} {{- if and (eq .ResourceType "image") (ne .MediaType.SubType "svg") -}} {{- $width = .Width -}} {{- $height = .Height -}} {{- $srcSet = partial "image-srcset.html" . -}} {{- end -}} {{- end -}} {{- if .IsBlock -}} {{- else -}} {{- end -}}
