{{- /* Returns a YAML front matter block describing a page for Markdown outputs. JSON strings are valid YAML. */ -}} {{- $lines := slice "---" (printf "title: %s" (jsonify .Title)) (printf "url: %s" (jsonify .Permalink)) -}} {{- with .Description }}{{ $lines = $lines | append (printf "description: %s" (jsonify .)) }}{{ end -}} {{- if not .Date.IsZero }}{{ $lines = $lines | append (printf "date: %s" (.Date.Format "2006-01-02T15:04:05Z07:00")) }}{{ end -}} {{- if and (not .Lastmod.IsZero) (ne .Lastmod .Date) }}{{ $lines = $lines | append (printf "lastmod: %s" (.Lastmod.Format "2006-01-02T15:04:05Z07:00")) }}{{ end -}} {{- with site.Params.author }}{{ $lines = $lines | append (printf "author: %s" (jsonify .)) }}{{ end -}} {{- with .Params.categories }}{{ $lines = $lines | append (printf "categories: %s" (jsonify .)) }}{{ end -}} {{- with .Params.tags }}{{ $lines = $lines | append (printf "tags: %s" (jsonify .)) }}{{ end -}} {{- $lines = $lines | append "---" -}} {{- return (delimit $lines "\n") -}}
