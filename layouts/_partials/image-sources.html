{{- /* Returns the elements for a , one per preferred format in site.Params.imageFormats (all but the last, which the itself uses). Returns an empty string when there is only one format, so callers can skip the wrapper. Takes a dict with "image" (the resource) and "sizes" (the sizes attribute, omitted when empty). */ -}} {{- $image := .image -}} {{- $sizes := "" -}} {{- with .sizes }}{{ $sizes = printf ` sizes="%s"` . }}{{ end -}} {{- $out := "" -}} {{- range $format := after 1 (partial "image-formats.html" $image) -}} {{- with partial "image-srcset.html" (dict "image" $image "format" $format) -}} {{- $out = printf `%s` $out $format . $sizes -}} {{- end -}} {{- end -}} {{- return $out -}}
