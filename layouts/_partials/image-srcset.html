{{- /* Builds a WebP srcset string for an image resource from site.Params.imageSizes. Widths larger than the original image are skipped so images are never upscaled; the original width is used instead when any were skipped. Returns an empty string when no sizes are configured. */ -}} {{- $image := . -}} {{- $entries := slice -}} {{- $widths := slice -}} {{- range site.Params.imageSizes -}} {{- $w := int . -}} {{- if gt $w $image.Width -}}{{- $w = $image.Width -}}{{- end -}} {{- if not (in $widths $w) -}} {{- $widths = $widths | append $w -}} {{- $url := ($image.Resize (printf "%dx webp" $w)).Permalink -}} {{- $entries = $entries | append (printf "%s %dw" $url $w) -}} {{- end -}} {{- end -}} {{- return (delimit $entries ",") -}}
